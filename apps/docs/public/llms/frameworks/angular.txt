# SSGOI with Angular

Create one SSGOI root above the router outlet.

```bash
npm install @ssgoi/angular
```

The inline shell style below uses modern `overflow-x:clip`. For older targets,
use `overflow-x:hidden` and add `min-height:100vh`.

```ts
import { Component, signal } from "@angular/core";
import { RouterOutlet } from "@angular/router";
import { Ssgoi, type SsgoiConfig } from "@ssgoi/angular";
import { drill } from "@ssgoi/angular/view-transitions";

@Component({
  selector: "app-root",
  standalone: true,
  imports: [RouterOutlet, Ssgoi],
  template: `
    <main
      ssgoi
      [config]="config()"
      style="position:relative;z-index:0;min-height:100dvh;overflow-x:clip"
    >
      <router-outlet />
    </main>
  `,
})
export class AppComponent {
  protected readonly config = signal<SsgoiConfig>({
    transitions: [
      { on: "/posts/**", except: "/posts", transition: drill() },
    ],
  });
}
```

## Page boundary

Mark the routed component root:

```html
<article data-ssgoi-transition="/posts/42">
  <!-- page -->
</article>
```

The id must match the config route patterns.

Angular may reuse the same routed component when only route params change.
Changing the attribute on the same DOM element does not create an OUT/IN pair.
For param-to-param transitions, force the marked root to remount:

```ts
// post-detail.component.ts
import { Component, inject } from "@angular/core";
import { toSignal } from "@angular/core/rxjs-interop";
import { ActivatedRoute } from "@angular/router";
import { map } from "rxjs";

@Component({
  standalone: true,
  template: `
    @for (transitionId of transitionIds(); track transitionId) {
      <article [attr.data-ssgoi-transition]="transitionId">
        <!-- detail page -->
      </article>
    }
  `,
})
export class PostDetailComponent {
  private readonly route = inject(ActivatedRoute);

  protected readonly transitionIds = toSignal(
    this.route.paramMap.pipe(
      map((params) => [`/posts/${params.get("id")}`]),
    ),
    { initialValue: [] as string[] },
  );
}
```
